{
    int i, j, pivot, temp;

    {
        pivot = a[low];
        i = low;
        j = high;
        while(i < j)
        {
            while(a[i] <= pivot && i < high)
                i++;
            while(a[j] > pivot)
                j--;
            if(i < j)

                a[i] = a[j];
                a[j] = temp;

        temp = a[low];
        a[low] = a[j];
        a[j] = temp;

#include <stdio.h>
void countSort(int a[], int n, int index, int count[], int max)
{
    if(index == n)
        return;
    count[a[index]]++;
    countSort(a, n, index + 1, count, max);
}
void displaySorted(int count[], int value, int max)
{
    if(value > max)
        return;
    if(count[value] > 0)
    {
        printf("%d ", value);
        count[value]--;
        displaySorted(count, value, max);
    }
    else
    {
        displaySorted(count, value + 1, max);
    }
}
int main()
{
    int a[100], count[100] = {0};
    int n, i, max;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter %d integers:\n", n);
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);
    max = a[0];
    for(i = 1; i < n; i++)
    {
        if(a[i] > max)
            max = a[i];
    }
    /* Recursive Counting Sort */
    countSort(a, n, 0, count, max);
    printf("Array in ascending order:\n");
    displaySorted(count, 0, max);
    return 0;

#include <stdio.h>
#include <ctype.h>
#define MAX 100
int stack[MAX];
int top = -1;
/* Push */
void push(int x)
{
    stack[++top] = x;
}
/* Pop */
int pop()
{
    return stack[top--];
}
/* Evaluate Postfix */
void evaluate(char exp[])
{
    int i, a, b, result;
    top = -1;
    for(i = 0; exp[i] != '\0'; i++)
    {
        if(isdigit(exp[i]))
        {


            push(exp[i] - '0');

        else
        {
            b = pop();
            a = pop();
